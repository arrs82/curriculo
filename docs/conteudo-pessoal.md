        sudo passwd root

        sudo apt-get update && sudo apt-get upgrade

        sudo apt-get dist-upgrade && sudo apt-get autoremove && sudo apt-get autoclean

        sudo ufw enable

        sudo ufw disable

        sudo ufw reload

        sudo ufw status verbose

        source ~/.bashrc
